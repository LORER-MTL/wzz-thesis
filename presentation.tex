\section{Background}

\begin{frame}{Convex optimization}
\begin{equation}\label{e:convex}
\begin{array}{ll}
\mbox{minimize}  &f_0(x) \\
\mbox{subject to} &f_i(x) \leq 0, \quad i=1, \ldots, m\\
&Ax=b
\end{array}
\end{equation}

\begin{minipage}{0.68\textwidth}
  \centering
  \begin{itemize}
    \item variable $x\in \reals^n$
    \item equality constraints are linear and $f_0, \ldots, f_m$ are convex
    \item can be solved globally and efficiently (both in theory and practice)
  \end{itemize}
\end{minipage}%
\vfill
\end{frame}

\begin{frame}{Functional form}
The user writes the problem in a general functional form:
\begin{equation}\label{e:functional}
\begin{array}{ll}
\mbox{minimize}   & f_0(x_1, \ldots, x_k) \\
\mbox{subject to} & f_i(x_1, \ldots, x_k) \leq g_i(x_1, \ldots, x_k), \quad i=1,\ldots,m_1\\
                  & h_i(x_1, \ldots, x_k) = j_i(x_1, \ldots, x_k), \quad i=1,\ldots,m_2
\end{array}
\end{equation}
\begin{columns}[T]
\begin{column}{0.58\textwidth}
\begin{itemize}
  \item encompasses any optimization problem
  \item potentially many variables of different sizes $x_1, \ldots, x_k$
  \item the functions are \emph{expression trees} built from a library of \emph{atoms}
\end{itemize}
\end{column}
\begin{column}{0.42\textwidth}
\centering\footnotesize example
\[
\begin{array}{ll}
\mbox{minimize}   & \|Fx - g\|_2 + \mu\|x\|_2 \\
\mbox{subject to} & x \succeq 0
\end{array}
\]
\end{column}
\end{columns}
\vfill
\end{frame}

\begin{frame}{Expression trees}
\tikzset{external/export=false}
Internally the functional form is stored as \textbf{expression trees} --- one per objective and
constraint. Atoms are the nodes; variables, constants, and parameters are the leaves.
\begin{center}
\begin{tikzpicture}
  [level 1/.style={sibling distance=30mm},
   level 2/.style={sibling distance=18mm}]
  \node {\annotatedop{\|\cdot\|_2}{\|Fx-g\|_2}}
    child {node {\annotatedop{-}{Fx-g}}
        child {node {\annotatedop{\texttt{matmul}}{Fx}}
              child {node[draw] {$F$}}
              child {node[circle, draw] {$x$}} }
        child {node[draw] {$g$}} };
\end{tikzpicture}
\end{center}
\end{frame}

\begin{frame}{Disciplined convex programming (DCP)}
\begin{itemize}
\item a grammar for specifying convex optimization problems [Grant et al., 2006]
\item convex and concave functions are constructed by combining a set of \textbf{atoms}
\item each atom has known curvature and monotonicity
\item simple rules specify how atoms can be composed
\item a problem that follows the DCP rules is convex-by-construction
\end{itemize}
\vfill
\end{frame}

\begin{frame}{Modeling languages implementing DCP}
\begin{itemize}
\item Yalmip (MATLAB), CVX (MATLAB), CVXPY (Python), Convex.jl (Julia), CVXR (R)
\item classic commercial software like AMPL and GAMS do not implement DCP
\pause
\bigskip
\item two main tasks of DCP-based modeling languages:
\begin{itemize}
  \item \textbf{analysis}: verify that a high level problem description follows the DCP rules
  \item \textbf{synthesis}: rewrite it into an equivalent conic standard form:
\end{itemize}
\end{itemize}
\[
\begin{array}{ll}
\mbox{minimize}   & c^\top x + d \\
\mbox{subject to} & b - Ax \in \mathcal{K}
\end{array}
\]
\centering{\footnotesize problem data $A, b, c, d$; \;$\mathcal K$ a product of standard cones}
\end{frame}

\begin{frame}{Functional conic form}
Canonicalization first rewrites the problem into a \emph{functional} conic form:
\begin{equation}\label{e:funcconic}
\begin{array}{ll}
\mbox{minimize}   & f_0(x_1, \ldots, x_k) \\
\mbox{subject to} & g_i(x_1, \ldots, x_k) \in \mathcal{K}_i, \quad i=1,\ldots,m
\end{array}
\end{equation}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{itemize}
  \item $f_0$ and every $g_i$ are now affine
  \item each $\mathcal K_i$ is a standard cone
  \item nonsmooth atoms are replaced by epigraph variables and cone constraints
  [Grant \& Boyd, 2008]
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\centering\footnotesize least squares becomes:
\[
\begin{array}{ll}
\mbox{minimize} & t_1 + \mu\, t_2\\
\mbox{s.t.} & x \in \reals_+^n\\
            & (t_1,\, Fx - g) \in \mathcal K_{\mathrm{soc}}\\
            & (t_2,\, x) \in \mathcal K_{\mathrm{soc}}
\end{array}
\]
\end{column}
\end{columns}
\vfill
\end{frame}

\begin{frame}{Quadratic conic standard form}
The backend materializes \eqref{e:funcconic} into the numeric conic standard form:
\begin{equation}\label{e:quadconic}
\begin{array}{ll}
\mbox{minimize}   & \tfrac12 x^\top P x + c^\top x + d \\
\mbox{subject to} & b - A x \in \mathcal{K}
\end{array}
\end{equation}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{itemize}\small
\item data $P \succeq 0,\ A, b, c, d$; \;$\mathcal K$ a product of standard cones
\item generalizes both QP ($P\neq0$, $\mathcal K=\reals^m_+$) and conic programs ($P=0$)
\item solvers: OSQP, HiGHS (QP); ECOS (conic); MOSEK, Clarabel (conic-quadratic)
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\centering\footnotesize least squares \eqref{e:funcconic}, $u = [\,x;\ t_1;\ t_2\,]$ ($P=0$):
\[
\min\; c^\top u, \quad b - A u \in \mathcal{K}
\]
{\scriptsize
\[
c = \begin{bmatrix} 0\\ 1\\ \mu \end{bmatrix},\;
A = \begin{bmatrix} -I & 0 & 0\\ 0 & -1 & 0\\ -F & 0 & 0\\ 0 & 0 & -1\\ -I & 0 & 0 \end{bmatrix},\;
b = \begin{bmatrix} 0\\ 0\\ -g\\ 0\\ 0 \end{bmatrix}
\]
\[
\mathcal K = \reals^n_+ \times \mathcal K_{\mathrm{soc}} \times \mathcal K_{\mathrm{soc}}
\]
}
\end{column}
\end{columns}
\vfill
{\scriptsize a squared objective $\|Fx-g\|_2^2$ would instead populate $P$, with no epigraph variable $t_1$.}
\end{frame}

\begin{frame}{Summary}
We have introduced four standard forms: the convex optimization standard form \eqref{e:convex},
the user's functional form \eqref{e:functional}, the functional conic form \eqref{e:funcconic},
and the numeric conic standard form \eqref{e:quadconic}.
\bigskip

A DCP modeling language works in two stages:
\begin{itemize}
\item the \emph{frontend} takes the functional form \eqref{e:functional}, uses DCP to verify it is
      convex \eqref{e:convex}, and rewrites it into the functional conic form \eqref{e:funcconic}
\item the \emph{backend} materializes \eqref{e:funcconic} into the numeric data \eqref{e:quadconic}
\end{itemize}
\pause
\bigskip
This thesis focuses on the \emph{canonicalization backend} for large-scale
and parametrized problems.
\end{frame}

\begin{frame}{Variable offsets}
\begin{itemize}
\item the backend stacks all variables into one vector $u$
\item each variable has a unique \textbf{id}; \texttt{id\_to\_col} maps it to a \textbf{column offset}
\item a variable of size $n$ occupies $n$ consecutive columns of $A$
\end{itemize}
\vspace{1ex}
\[
x \in \reals^2,\quad y \in \reals^3 \qquad\Rightarrow\qquad
u = \big[\,\underbrace{x_1\ x_2}_{\text{cols }1\text{--}2}\ \ \underbrace{y_1\ y_2\ y_3}_{\text{cols }3\text{--}5}\,\big]^\top \in \reals^5
\]
\[
\texttt{id\_to\_col} = \{\, \mathrm{id}(x)\!: 0,\ \ \mathrm{id}(y)\!: 2 \,\}
\]
\begin{itemize}
\item so $A$ has $5$ columns (\texttt{var\_length}); a final column holds the constant $b$
\end{itemize}
\vfill
\end{frame}

\section{A lightweight C engine}

\begin{frame}{The key insight}
\begin{itemize}
\item for any parameterized problem, under mild conditions the \textbf{sparsity pattern} of $A(\theta)$ is fixed across the family
\item only the \textbf{numerical values} of the nonzeros change with $\theta$
\pause
\item reframe matrix stuffing as \textbf{sparse Jacobian evaluation}:
  \begin{itemize}
    \item symbolic phase: compute the sparsity pattern of $\partial A / \partial \theta$ once
    \item numeric phase: refill values on every solve
  \end{itemize}
\item no restriction on how parameters enter the problem
\end{itemize}
\vfill
\end{frame}

\begin{frame}{Two-phase design}
\textbf{Init phase} (run once per problem family):
\begin{itemize}
\item allocate one \texttt{expr} struct per tree node
\item traverse the tree to compute sparsity patterns
\item allocate Jacobians as CSR matrices
\end{itemize}
\vspace{1ex}
\textbf{Eval phase} (run on every solve, with new $\theta$):
\begin{itemize}
\item walk the tree, fill numerical values into pre-allocated buffers
\item stuff the resulting $A$, $b$, $c$ into the solver
\end{itemize}
\pause
\vspace{1ex}
Cost model: init is $O(\text{tree size})$ once; eval is $O(\text{nnz}(A))$ per solve.
\vfill
\end{frame}

\begin{frame}{The two phases on our example}
replace the constant $3$ with a parameter $\theta$: \quad $x + \theta y$, \; $x,y,z \in \reals^2$
\vspace{2ex}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{Init} (once per problem family)
\begin{itemize}
\item walk the tree, record \emph{which} blocks are nonzero
\item pattern: blocks $x, y$ \;($z$ empty)
\item allocate the CSR matrix $\begin{bmatrix} \bullet & \bullet & \plotzero \end{bmatrix}$
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{Eval} (every solve, new $\theta$)
\begin{itemize}
\item refill values into the fixed pattern
\item $\theta{=}3:\ \begin{bmatrix} I & 3I & \plotzero \end{bmatrix}$
\item $\theta{=}7:\ \begin{bmatrix} I & 7I & \plotzero \end{bmatrix}$
\item structure never recomputed
\end{itemize}
\end{column}
\end{columns}
\pause
\vspace{2ex}
the current backend redoes the \textbf{whole} left column on every solve; the engine pays it once.
\par\smallskip
\footnotesize (quadratic objectives add a matrix $P$, built the same way from second-order coefficients.)
\end{frame}

\begin{frame}[fragile]{The \texttt{expr} struct}
\begin{lstlisting}[language=C]
typedef struct expr {
    int    d1, d2, size;    /* shape (d1 x d2) */
    int    n_vars;
    int    var_id;          /* offset, or NOT_A_VARIABLE */

    struct expr *left, *right;

    double     *value;      /* forward-pass output */
    CSR_Matrix *jacobian;   /* size x n_vars sparse */

    forward_fn       forward;
    jacobian_init_fn jacobian_init_impl;
    eval_jacobian_fn eval_jacobian;
} expr;
\end{lstlisting}
\vfill
\end{frame}

\begin{frame}{The atom contract}
Every atom implements three functions:
\vspace{1ex}
\begin{center}
\begin{tabular}{ll}
\toprule
\textbf{Function} & \textbf{Phase / Purpose} \\
\midrule
\texttt{forward}            & numeric: compute the atom's output value \\
\texttt{jacobian\_init}     & symbolic: allocate sparsity pattern of the Jacobian \\
\texttt{eval\_jacobian}     & numeric: fill Jacobian values from children \\
\bottomrule
\end{tabular}
\end{center}
\vspace{1ex}
\begin{itemize}
\item \texttt{jacobian\_init} is called \textbf{once} during the init phase
\item \texttt{eval\_jacobian} is called on \textbf{every} solve
\end{itemize}
\vfill
\end{frame}

\begin{frame}{Atom example: elementwise multiply}
\begin{itemize}
\item operation: $z = a \odot b$ (elementwise product of two expressions)
\item Jacobian: $\partial z / \partial x = \mathrm{diag}(a) \, \partial b / \partial x + \mathrm{diag}(b) \, \partial a / \partial x$
\pause
\item \texttt{jacobian\_init}: union of the sparsity patterns of $\partial a / \partial x$ and $\partial b / \partial x$
\item \texttt{eval\_jacobian}: scale child Jacobian rows by the current values of $a$ and $b$
\pause
\item key observation: the sparsity union is computed \textbf{once}; per-solve work is just row scaling
\item this is where \textbf{parameter values} enter without changing structure
\end{itemize}
\vfill
\end{frame}

\begin{frame}{Atom example: matrix multiplication}
\begin{itemize}
\item operation: $z = A \cdot y$ (where $A$ is a constant or parameter, $y$ is an expression)
\item Jacobian: $\partial z / \partial x = A \cdot \partial y / \partial x$
\pause
\item \texttt{jacobian\_init}: sparsity pattern of $A \cdot J_y$ derived from the sparsity of $A$ and $J_y$
\item \texttt{eval\_jacobian}: sparse-sparse multiply, writing into pre-allocated CSR buffers
\item the structural pattern --- which entries are nonzero --- is fixed once $A$'s sparsity is known
\end{itemize}
\vfill
\end{frame}

\begin{frame}{Assembling $A$}
\begin{itemize}
\item each constraint contributes a row block of $A$
\item after \texttt{eval\_jacobian} runs on every node, the root Jacobians \emph{are} the rows of $A$
\item we copy them into a flat CSR buffer in the layout the solver expects
\pause
\item the column structure of $A$ is computed once; row values are refilled per solve
\item no Python in the hot path; no allocation in the eval phase
\end{itemize}
\vfill
\end{frame}

\begin{frame}{What ``lightweight'' means}
\begin{itemize}
\item pure C; no Python in the inner loop
\item no symbolic graph mutation between solves
\item CSR storage throughout; cache-friendly traversals
\item no general-purpose AD framework (no JAX, no Autograd) --- we only implement the atoms we need
\pause
\item contrast with today's backend:
  \begin{itemize}
    \item no 3D parameter tensor in memory
    \item sparsity computed once, not on every solve
    \item no restriction on how parameters enter
  \end{itemize}
\end{itemize}
\vfill
\end{frame}

\section{Conclusions}

\begin{frame}{Summary}
\begin{itemize}
\item reframed CVXPY matrix stuffing as \textbf{sparse Jacobian evaluation}
\item built a lightweight \textbf{C engine} that:
  \begin{itemize}
    \item computes sparsity once per problem family
    \item refills values per solve
    \item avoids the 3D parameter tensor of the DPP path
  \end{itemize}
\item works for parameterized problems \textbf{without the DPP rule restriction}
\pause
\item (pending) benchmark wins over ignore\_DPP, competitive with DPP
\end{itemize}
\vfill
\end{frame}

\begin{frame}{Limitations and future work}
\textbf{Limitations}
\begin{itemize}
\item partial atom coverage --- not yet a drop-in replacement for CVXPY
\item C library, not yet integrated as a CVXPY backend
\item single-threaded; no GPU
\end{itemize}
\vspace{1ex}
\textbf{Future work}
\begin{itemize}
\item expand atom library to cover all of CVXPY's standard atoms
\item ship as a Python-importable CVXPY backend
\item explore non-DPP parameterizations where sparsity is still stable
\end{itemize}

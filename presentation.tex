\begin{frame}{Convex optimization}
\vspace{-1ex}
\[
  \begin{array}{ll}
    \underset{\mvec{x}}{\mbox{minimize}} & f_0(\mvec{x}) \\[0.5ex]
    \mbox{subject to}                    & f_i(\mvec{x}) \le 0 \\
                                         & g_j(\mvec{x}) = 0.
  \end{array}
\]

\vspace{1ex}
\begin{itemize}\small\bul
  \item variable $\mvec{x} \in \reals^n$;
  \item the equality constraints $g_j$ are affine, and $f_0$ and the $f_i$ are convex;
  \item why care about convexity?
    \begin{itemize}\small
      \item<2->[--] classic answer: a local optimum is a global optimum;
      \item<2->[--] modern answer: very actionable, because of reliable software.
    \end{itemize}
\end{itemize}

\vfill
{\scriptsize\color{black!55} Boyd and Vandenberghe (2004).}
\end{frame}

% --- I1c ------------------------------------------------------------------------
%  The form every general-purpose solver reads, shown before the zoom I1 draws it as a box.
%  Source: 5-Introduction.tex:63-81 (eq:quad-conic-form); cones listed at :278-283.
%  If asked which problems this covers: K = R^m_+ gives a QP, P = 0 a standard conic program
%  (5-Introduction.tex:78-80). Bullet 3 is 5-Introduction.tex:35-37 (bental2001lectures).
\begin{frame}{Conic form}
\vspace{-1ex}
\[
  \begin{array}{ll}
    \underset{\mvec{x} \in \reals^n}{\mbox{minimize}}
      & \frac{1}{2}\mvec{x}^\top \mmat{P} \mvec{x} + \mvec{c}^\top \mvec{x} \\[0.6ex]
    \mbox{subject to} & \mvec{b} - \mmat{A}\mvec{x} \in \cone{K}.
  \end{array}
\]

\vspace{1ex}
\begin{itemize}\small\bul
  \item problem data: $\mmat{P} \in \symm^n_+$,
        $\mmat{A} \in \reals^{m \times n}$, $\mvec{b} \in \reals^m$, $\mvec{c} \in \reals^n$;
  \item $\cone{K} \subseteq \reals^m$ is a product of standard cones;
  \item all general-purpose solvers target this standard form.
\end{itemize}

\vfill
{\scriptsize\color{black!55} Ben-Tal and Nemirovski (2001).}
\end{frame}

\begin{frame}{Disciplined convex programming}
\vspace{-0.5ex}
\begin{itemize}\small\bul
  \item a grammar for specifying convex optimization problems;
  \item convex and concave functions are built by combining a set of \textbf{atoms};
  \item each atom has known curvature, monotonicity, and sign;
  \item simple rules specify how atoms can be composed.
\end{itemize}

\vspace{2.5ex}
\onslide<2->{%
\begin{itemize}\small\bul
  \item a problem that follows the DCP rules is {\color{polyblue}convex by construction};
  \item and can be solved globally and efficiently --- in theory and in practice.
\end{itemize}}

\vfill
{\scriptsize\color{black!55} Grant, Boyd and Ye (2006).}
\end{frame}

\begin{frame}{Example: fitting with regularization}
\recallstrip[real world]{1}

\vspace{2ex}
\[
  \begin{array}{ll}
    \mbox{minimize}   & \|\mmat{F}\mvec{x}-\mvec{g}\|_2 + \mu \|\mvec{x}\|_2 \\[0.3ex]

  \end{array}

        parameters $\mvec{g} \in \reals^m$ and $\mu \ge 0$;
  \item similar to least-squares;
  \item physical constraint such as non-negativity of the solution.
\end{itemize}
\end{frame}

\begin{frame}{Canonicalization: the frontend}
\recallstrip{2}
\vspace{-1ex}
\[
  \begin{array}{ll}
    \underset{t_1,\, t_2,\, \mvec{x}}{\mbox{minimize}} & t_1 + \mu t_2 \\[0.4ex]
    \mbox{subject to}
      & \mvec{x} \ge \zeros \\
      & (t_1,\ \mmat{F}\mvec{x} - \mvec{g}) \in \cone{K}_{\mathrm{soc},\, m+1} \\
      & (t_2,\ \mvec{x}) \in \cone{K}_{\mathrm{soc},\, n+1}.
  \end{array}
\]

\vspace{-0.5ex}
\begin{center}
{\footnotesize\color{black!55}
$\cone{K}_{\mathrm{soc},\,k}
   = \{(u,\mvec{v}) \in \reals \times \reals^{k-1} : \|\mvec{v}\|_2 \le u\}$,
\qquad
$\cone{K} = \reals_+^n \times \cone{K}_{\mathrm{soc},\, m+1}
            \times \cone{K}_{\mathrm{soc},\, n+1}$}
\end{center}

\vspace{1ex}
\begin{itemize}\small\bul
  \item the nonlinear norms are replaced by new variables $t_1$, $t_2$;
  \item {\color{polyblue}every constraint is an affine function belonging to a cone.}
\end{itemize}
\end{frame}

% --- B-DCP ------------------------------------------------------------------------
%  Q: "what exactly are the DCP rules?"  All three clauses, compliance, and the caveat.
%  Source: 5-Introduction.tex:227-250 (rules) and :178-197 (atom table, cut to five rows).
%  Shares the table with F3e on purpose: F3e answers "what is an atom?".
\begin{frame}[label=bk-dcp]{The DCP rules}
\vspace{-0.5ex}
{\small Every atom declares its curvature, its monotonicity in each argument, and the sign of
its output:}
\vspace{0.5ex}
\begin{center}\footnotesize
\begin{tabular}{l l l l l}
\toprule
Atom & Definition & Curvature & Monotonicity & Sign \\
\midrule
\texttt{sum}    & $\textstyle\sum_i x_i$ & affine  & non-decreasing & any \\
\texttt{abs}    & $|x|$                  & convex  & unknown        & nonnegative \\
\texttt{square} & $x^2$                  & convex  & unknown        & nonnegative \\
\texttt{sqrt}   & $\sqrt{x}$             & concave & non-decreasing & nonnegative \\
\texttt{norm}   & $\|\mvec{x}\|_2$       & convex  & unknown        & nonnegative \\
\bottomrule
\end{tabular}
\end{center}

\vspace{1ex}
{\small The composition $\phi(u_1(\mvec{x}), \dots, u_k(\mvec{x}))$ is
{\color{polyblue}convex} when $\phi$ is convex and, for {\color{polyblue}each} argument $i$,
one of the following holds:}
\vspace{-0.5ex}
\begin{itemize}\small
  \item $\phi$ is non-decreasing in $i$ and $u_i$ is convex;
  \item $\phi$ is non-increasing in $i$ and $u_i$ is concave; or
  \item $u_i$ is affine.
\end{itemize}
\vspace{0.5ex}
{\small Swap convex and concave for the concave rule. A problem is DCP-compliant when the
objective and every constraint come out compatible: convex objective,
convex $\le$ concave, affine equalities.}

\vspace{1ex}
\begin{center}\small
The rules are sufficient for convexity, not necessary: a genuinely convex problem written in an
unrecognized form is rejected.
\end{center}
\end{frame}

% --- B-NONDCP ---------------------------------------------------------------------
%  Q: "give me a problem the rules reject."  The previous slide's rules on sqrt(x^2+1); the
%  rewrite is the SOC form of the norm atom, 5-Introduction.tex:283-288.
\begin{frame}[fragile,label=bk-nondcp]{Sufficient, not necessary}
\vspace{-0.5ex}
{\small The same function, written two ways. Only one of them compiles.}
\vspace{0.5ex}
\[
  \sqrt{x^2 + 1} \;=\; \bigl\| (x,\ 1) \bigr\|_2
\]
\vspace{-2ex}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{lstlisting}[language=mypython, basicstyle=\ttfamily\scriptsize,
                   aboveskip=0.4em, belowskip=0.4em]
cp.sqrt(cp.square(x) + 1)
\end{lstlisting}
\vspace{0.5ex}
{\footnotesize\color{polyblue}\textbf{rejected}}
\begin{itemize}\footnotesize
  \item \texttt{sqrt} is concave and non-decreasing
  \item its argument $x^2 + 1$ is convex, not affine
  \item no rule covers concave $\circ$ convex, so the curvature is unknown
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\begin{lstlisting}[language=mypython, basicstyle=\ttfamily\scriptsize,
                   aboveskip=0.4em, belowskip=0.4em]
cp.norm(cp.hstack([x, 1]))
\end{lstlisting}
\vspace{0.5ex}
{\footnotesize\color{polyblue}\textbf{accepted}}
\begin{itemize}\footnotesize
  \item \texttt{norm} is convex
  \item its argument $(x, 1)$ is affine
  \item convex $\circ$ affine is convex, and the epigraph is
        $(t,\ x,\ 1) \in \cone{K}_{\mathrm{soc},3}$
\end{itemize}
\end{column}
\end{columns}

\vspace{1.5ex}
\begin{center}\small
The rules are a test on the expression, not on the function. Rewriting is the user's job; the
atom library is what makes it possible.
\end{center}
\end{frame}

% --- B-DPP ------------------------------------------------------------------------
%  Q: "what is DPP, and what does CVXPY do instead?"  The grammar, the cached maps with their
%  sizes, and the three backends; the baseline the parametric results are measured against.
%  Sources: 6-Revue_litterature.tex:376-397 (the grammar, eq:param-maps, the sizes of Q and R),
%           :423-431 (ignore_dpp), :458-470 (the three backends), :478-479 (default selection).
%  Spoken: "a re-solve is then one sparse matrix-vector product per map."
\begin{frame}[label=bk-dpp]{DPP and the parameter maps}
\vspace{-0.5ex}
{\small DPP is a second grammar on top of DCP: parameters enter affinely, and no product of two
parametrized expressions is allowed. The problem data is then
{\color{polyblue}affine in the parameters}:}
\[
  \mvec{c} = \mmat{Q}\,\gvec{\tilde\theta},
  \qquad
  \vect([\mmat{A}\ \mvec{b}]) = \mmat{R}\,\gvec{\tilde\theta},
  \qquad
  \gvec{\tilde\theta} = \begin{bmatrix} \gvec{\theta} \\ 1 \end{bmatrix}
\]
\vspace{-1ex}
\begin{center}\small
$\mmat{Q} \in \reals^{n \times (p+1)}, \quad \mmat{R} \in \reals^{m(n+1) \times (p+1)}$
\quad{\color{black!55}($n$ variables, $m$ rows, $p$ parameter entries)}
\end{center}

\begin{itemize}\small\bul
  \item built by giving every coefficient matrix of the pass a third dimension, indexed by the
        parameter entries; a re-solve is then one sparse matrix-vector product per map
  \item three backends, three representations of the same tensor:
    \begin{itemize}\footnotesize
      \item[--] \texttt{cvxcore} (C++): one sparse matrix per parameter
      \item[--] \texttt{SCIPY}: the slices stacked into one strided sparse matrix
      \item[--] \texttt{COO}: never formed, triples with a parameter index
    \end{itemize}
  \item outside the grammar, or with \texttt{ignore\_dpp=True} (as in the main benchmark): no
        tensor, and the whole canonicalization repeats
\end{itemize}

\vfill
% agrawal2019dpp: DPP is named on three frames, so the References list must back it.
{\scriptsize\color{black!55} Agrawal et al. (2019).}
\end{frame}

\begin{frame}[label=bk-families]{Two ways to get the coefficients}
\vspace{-0.5ex}
{\small Every modelling language reviewed produces the same output, the solver problem data. They
split into two families by what they ask of the user.}

\vspace{1.5ex}
{\small\color{polyblue}\textbf{Differentiate an expression graph}}
\begin{itemize}\small
  \item \texttt{CasADi}: built for nonlinear optimization; its conic interface \texttt{qpsol}
        differentiates the graph through matrix-vector products, so the sparsity pattern must be
        {\color{polyblue}detected at runtime} and compressed by colouring
  \item \cvxpy{}: composes the linear operators of the conic-form trees in one bottom-up pass,
        which is equivalent to differentiating them; expressions stay vectorized, so no detection
        step is needed
  \item {\color{polyblue}\sde{}}: differentiates them outright, one Jacobian per node with
        respect to \emph{all} the variables, so the chain rule is a single matrix product ---
        not a dictionary of blocks keyed by variable
\end{itemize}
\end{frame}

\begin{frame}{Two ways to get the coefficients (continued)}
\vspace{-0.5ex}
{\small\color{polyblue}\textbf{Read the coefficients the user declares}}
\begin{itemize}\small
  \item \texttt{AMPL}, \texttt{JuMP}: the user writes each objective and constraint as a sum of
        terms, and every linear and quadratic coefficient is stored as it is typed
  \item neither has an atom library or a canonicalization step, so a norm is left to the solver
        (\texttt{AMPL}) or written in conic form by hand (\texttt{JuMP})
  \item both scalarize their models
\end{itemize}
\end{frame}

\begin{frame}[label=bk-matfree]{Why not matrix-free?}
\vspace{-0.5ex}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.47\textwidth}
{\small\color{polyblue}\textbf{Matrix-free}}
\begin{itemize}\footnotesize\bul
  \item every linear function is an abstract operator that only applies $\mmat{A}\mvec{x}$
        and $\mmat{A}^\top\mvec{y}$
  \item no assembly, so no assembly cost
  \item serves first-order and Krylov-based solvers, which need nothing more than those
        products
\end{itemize}
\end{column}
\begin{column}{0.47\textwidth}
{\small\color{polyblue}\textbf{Explicit matrix}}
\begin{itemize}\footnotesize\bul
  \item \texttt{Clarabel} and \texttt{MOSEK} (interior point), \texttt{SCS} and \texttt{OSQP}
        in their default direct mode all factorize a matrix built from $\mmat{A}$ and
        $\mmat{P}$
  \item a factorization needs the entries, so these solvers need $\mmat{A}$ formed
  \item this thesis keeps $\mmat{A}$ explicit and makes it cheaper to form
\end{itemize}
\end{column}
\end{columns}

\vspace{2ex}
\begin{center}\small\color{polyblue}
Matrix-free removes the cost for the solvers that can do without $\mmat{A}$; the solvers
\cvxpy{} calls by default cannot.
\end{center}

% --- B1 ---------------------------------------------------------------------------
%  Q: "permuted dense cannot be closed under addition, surely?"  Why SC3's `add' row runs on
%  CSR, plus what the row omits: the views alias the dense blocks. Source: 7-Theme1.tex:325-331.
\begin{frame}[label=bk-addition]{Addition is where the dense path stops}
\vspace{-0.5ex}
\begin{itemize}\small\bul
  \item the Jacobian of $\mmat{F}\mvec{x} + \mvec{y}$ holds the block of $\mmat{F}$ at the
        columns of $\mvec{x}$ \emph{and} an identity at the columns of $\mvec{y}$
  \item no single triple $(\mmat{X}, \mvec{r}, \mvec{c})$ covers both, so the sum is performed
        on CSR views of the two operands instead
  \item the views are built during the symbolic pass and their value arrays point at the dense
        blocks, so {\color{polyblue}no values are copied}
  \item the result is a sparse matrix, and so is everything computed from it
\end{itemize}

\vspace{2ex}
\begin{center}\small
Products, by contrast, keep the dense representations: a product with a permuted dense or
stacked permuted dense operand gives back a matrix of the same kind.
\end{center}
\end{frame}

% --- B-PROFILE -----------------------------------------------------------------------

\begin{frame}[label=bk-profile]{How to read a performance profile}
\vspace{-0.5ex}
{\small A performance profile compares the backends over the whole suite at once, rather than
one problem at a time.}

\vspace{2ex}
\[
  f_s(\tau) = \frac{1}{n}\,\bigl|\{\, p : t_{p,s} \le \tau \textstyle\min_b t_{p,b}
  \,\}\bigr|
\]

\vspace{1ex}
\begin{itemize}\small\bul
  \item $t_{p,s}$ is the time of backend $s$ on problem $p$, and $\min_b t_{p,b}$ is the time
        of the fastest backend on that same problem
  \item $f_s(\tau)$ is the fraction of the $n$ problems on which backend $s$ lands within a
        factor $\tau$ of that best time
  \item {\color{polyblue}at $\tau = 1$}: the fraction of problems on which the backend is
        outright fastest
  \item {\color{polyblue}where the curve reaches $1$}: the factor the backend needs in order to
        cover every problem in the suite
\end{itemize}

\vspace{1.5ex}
{\scriptsize\color{black!55} Dolan and Mor\'e (2002).}
\end{frame}

%  Q: "show me a problem you lose."  SDPSegfault1132, not ConvexPlasticity: ten lines, a
%  recognizable SDP, visibly dense constants. Name is the suite's own (cvxpy issue 1132).
%  Code trimmed from cvxpy/benchmarks benchmark/sdp_segfault_1132_benchmark.py (D, V shown as

%  Sources: 7-Theme2.tex:56-59, tab:oneshot-compile (7-Theme2.tex:119) for the timings.
%  Cancellation and zero-dropping: plans/sdpsegfault_analysis_20260822.md (`Deeper dig'):
%  49.5 M -> 1.01 M nnz, measured; SCIPY's csr addition prunes exact zeros at the `+', the
%  engine carries 48.5 M exact zeros to cone formatting (~15 s tail vs 0.1 s). The cancellation
%  is the Gram-to-distance identity D_ij = G_ii + G_jj - 2 G_ij. Measured on the dev build.
\begin{frame}[fragile,label=bk-loss]{SDPSegfault1132: the code}
\vspace{-1ex}
{\scriptsize\color{black!55} \texttt{SDPSegfault1132}, where \de{} is
$1.39\times$ behind}

\begin{center}
\begin{minipage}{0.82\textwidth}
\begin{lstlisting}[language=mypython, basicstyle=\ttfamily\small,
                   aboveskip=0.2em, belowskip=0.2em]
n = 100
V = ...                      # dense, (n, n-1)
D = ...                      # dense, (n, n)
W = np.ones((n, n))          # dense, (n, n)

G = cp.Variable((n-1, n-1), PSD=True)
A = cp.kron(e,   cp.reshape(cp.diag(V @ G @ V.T), (1, n)))
B = cp.kron(e.T, cp.reshape(cp.diag(V @ G @ V.T), (n, 1)))
C = cp.norm(cp.multiply(W, A + B - 2 * V @ G @ V.T - D), p='fro')

prob = cp.Problem(cp.Maximize(cp.trace(G) - C))
\end{lstlisting}
\end{minipage}
\end{center}

\begin{itemize}\small\bul
  \item every constant is dense, and \texttt{V @ G @ V.T} sandwiches the variable between
        two of them
  \item each \texttt{V @ G @ V.T} is a dense block of $49.5$\,M entries, but in the sum
        {\color{polyblue}almost all of them cancel}: $1.01$\,M nonzeros are left
  \item \texttt{SCIPY} treats everything as sparse and drops the zeros as it adds; \de{} keeps
        the dense blocks, zeros included, to the end: $13.21$ s against $9.51$ s
\end{itemize}
\end{frame}
